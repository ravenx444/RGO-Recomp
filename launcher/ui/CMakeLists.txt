cmake_minimum_required(VERSION 3.24)
project(JfgUiPreview LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature ZLIB BZIP2 PNG HARFBUZZ BROTLI)
 set(FT_DISABLE_${feature} ON CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(freetype GIT_REPOSITORY https://github.com/freetype/freetype.git GIT_TAG 526ec5c47b9ebccc4754c85ac0c0cdf7c85a5e9b)
FetchContent_MakeAvailable(freetype)
if(NOT TARGET Freetype::Freetype)
 add_library(Freetype::Freetype ALIAS freetype)
endif()
set(RMLUI_SHELL ON)
set(RMLUI_BACKEND Win32_DX11 CACHE STRING "" FORCE)
set(RMLUI_SAMPLES OFF CACHE BOOL "" FORCE)
set(RMLUI_COMPILER_OPTIONS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(rmlui GIT_REPOSITORY https://github.com/mikke89/RmlUi.git GIT_TAG ba95ffe8bfb6370efb2cdcca927eaad4710c5413)
FetchContent_MakeAvailable(rmlui)
# RGO: SDL2's GameController API discovers DualShock 4, DualSense, Switch Pro
# and Xbox controllers. Same release as the game's SDL2.dll (RT64's bundled
# SDL 2.26.3), so the launcher and the game number controllers alike.
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_LIBC ON CACHE BOOL "" FORCE)
set(SDL2_DISABLE_SDL2MAIN ON CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
set(SDL2_DISABLE_UNINSTALL ON CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
FetchContent_Declare(sdl2 URL https://github.com/libsdl-org/SDL/releases/download/release-2.26.3/SDL2-2.26.3.tar.gz
 URL_HASH SHA256=c661205a553b7d252425f4b751ff13209e5e020b876bbfa1598494af61790057)
FetchContent_MakeAvailable(sdl2)
if(NOT JFG_UI_NATIVE_ONLY)
add_executable(JFG-UiPreview WIN32 preview.cpp)
target_compile_features(JFG-UiPreview PRIVATE cxx_std_20)
target_compile_definitions(JFG-UiPreview PRIVATE UNICODE _UNICODE)
target_compile_options(JFG-UiPreview PRIVATE /utf-8 /W4)
target_link_libraries(JFG-UiPreview PRIVATE RmlUi::RmlUi rmlui_backend_Win32_DX11 xinput9_1_0 shell32 comdlg32)
add_custom_command(TARGET JFG-UiPreview POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:JFG-UiPreview>/ui"
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/fonts" "$<TARGET_FILE_DIR:JFG-UiPreview>/ui/fonts")

add_custom_command(TARGET JFG-UiPreview POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy "${rmlui_SOURCE_DIR}/LICENSE.txt" "$<TARGET_FILE_DIR:JFG-UiPreview>/ui/RmlUi-LICENSE.txt"
 COMMAND ${CMAKE_COMMAND} -E copy "${freetype_SOURCE_DIR}/docs/FTL.TXT" "$<TARGET_FILE_DIR:JFG-UiPreview>/ui/FreeType-LICENSE.txt")

endif()

if(JFG_UI_NATIVE_ONLY)
 add_library(jfg_frontend_ui STATIC native_ui.cpp live_tool_window.cpp)
 # Keep About tied to the launcher version and source that produced this binary.
 file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../windows/Launcher.cs" jfg_version_line REGEX "AssemblyInformationalVersion")
 string(REGEX REPLACE ".*Version\\(\"([^\"]+)\"\\).*" "\\1" JFG_UI_VERSION "${jfg_version_line}")
 execute_process(COMMAND git rev-parse --short=7 HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
     OUTPUT_VARIABLE JFG_UI_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
 if(NOT JFG_UI_REVISION)
  set(JFG_UI_REVISION "Source archive")
 endif()
 execute_process(COMMAND git status --porcelain --untracked-files=normal WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
     OUTPUT_VARIABLE jfg_ui_changes OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
 if(jfg_ui_changes)
  string(APPEND JFG_UI_REVISION " + local edits")
 endif()
 string(TIMESTAMP JFG_UI_BUILT "%Y-%m-%d UTC" UTC)
 configure_file(build_info.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/frontend_build_info.hpp" @ONLY)
 string(REGEX MATCH "^([0-9]+)\.([0-9]+)\.([0-9]+)" jfg_version_triplet "${JFG_UI_VERSION}")
 set(JFG_UI_VERSION_QUAD "${CMAKE_MATCH_1},${CMAKE_MATCH_2},${CMAKE_MATCH_3},0")
 configure_file(version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
 file(READ "${CMAKE_CURRENT_BINARY_DIR}/version.rc" jfg_version_resource)
 target_include_directories(jfg_frontend_ui PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

 target_include_directories(jfg_frontend_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
 target_compile_features(jfg_frontend_ui PRIVATE cxx_std_20)

 target_compile_options(jfg_frontend_ui PRIVATE /utf-8 /W4)
 target_link_libraries(jfg_frontend_ui PRIVATE RmlUi::RmlUi rmlui_backend_Win32_DX11 SDL2-static dcomp d3d11 dxgi xinput9_1_0 comdlg32 shell32)
 target_compile_definitions(jfg_frontend_ui PRIVATE SDL_MAIN_HANDLED)
 set(ui_rc "1 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico\"\n")
 set(id 201)
 foreach(asset assets/main.rml assets/theme.rcss fonts/Barlow-Regular.ttf fonts/Barlow-SemiBold.ttf fonts/ChakraPetch-Bold.ttf assets/live-map.rml assets/live-inventory.rml assets/live-tools.rcss assets/about.rml assets/about.rcss)
  string(APPEND ui_rc "${id} RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/${asset}\"\n")
  math(EXPR id "${id}+1")
 endforeach()
 string(APPEND ui_rc "${jfg_version_resource}")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ui.rc" "${ui_rc}")
 set(JFG_UI_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/ui.rc" PARENT_SCOPE)
endif()
